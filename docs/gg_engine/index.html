05 · Code snippets (optional)

Click the COPY button and paste into PowerShell on Windows. Each command knows the folder path; you do not need to remember cd. The simplest option is still to use the direct links above.

Download and open the ZIP without GitWorks with PowerShell; automatically creates a Downloads folder.
# One-time download: requires no Git, no cd
$folder = Join-Path $env:USERPROFILE "Downloads\gg_engine_W10"
New-Item -ItemType Directory -Force -Path $folder | Out-Null
$url = "https://raw.githubusercontent.com/GBOGEB/orchestration-sandbox/f13d993db7e8e7dd90ecbbe3789010b0e0b0d3c7/dist/gg_engine_w10_source.zip"
$zip = Join-Path $folder "gg_engine_w10_source.zip"
Invoke-WebRequest -Uri $url -OutFile $zip
Expand-Archive -Path $zip -DestinationPath (Join-Path $folder "source") -Force
Start-Process $folder

Complete Git handover development loopRequires Git and Python installed; never force-resets or merges.
# Optional developer route: no manual cd needed
$repo = Join-Path $env:USERPROFILE "Documents\gg_engine_w10"
$branch = "handover/gg-engine-w10-semantic-contract-v0p2"
if (-not (Test-Path (Join-Path $repo ".git"))) {
  git clone --single-branch --branch $branch "https://github.com/GBOGEB/orchestration-sandbox.git" $repo
} else {
  git -C $repo pull --ff-only
}
if ($LASTEXITCODE -ne 0) { throw "Git command failed" }
Push-Location $repo
try {
  py -3 -m unittest discover -s gg_engine_contract -p "test_*.py" -v
  if ($LASTEXITCODE -ne 0) { throw "Tests failed" }
  py -3 gg_engine_contract/package_release.py --output "dist/gg_engine_w10_source.zip"
  if ($LASTEXITCODE -ne 0) { throw "ZIP generator failed" }
  Get-FileHash "dist/gg_engine_w10_source.zip" -Algorithm SHA256
  Start-Process (Resolve-Path "dist").Path
} finally { Pop-Location }
